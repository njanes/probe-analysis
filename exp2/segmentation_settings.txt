# Fiji settings tuned for exp2: bead_segmentation.ijm opens with these when this folder is chosen
# (names as in results/settings_used.txt, which records what each run used).

# (16 of its 24 images found no bead). At 150 every image finds beads (100 beads instead of 51); the earlier
# run keeps 207 of the 221 beads it finds at 80.

shrinkPx=5
minThreshold=150
