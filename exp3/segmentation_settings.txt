# Fiji settings tuned for exp3: bead_segmentation.ijm opens with these when this folder is chosen
# (names as in results/settings_used.txt, which records what each run used).
# Lowest allowed threshold 150: the probe run's transmitted light was mostly taken at 530 to 570 V, so its
# images are grainy and at 80 the grain is outlined instead of the beads (most of its images found no bead). At 150 every image finds beads; the media run keeps 196 of the 206 beads it
# finds at 80. The many small round objects left out (about 3 um across, not fluorescent) are other particles,
# not beads.
minDiam=3.8
maxDiam=7.0
minCirc=0.75
maxCentreContrast=0.2
maxBrightSpot=0.35
minEdgeSharpness=0.09
shrinkPx=5
minThreshold=150
# Figure images (display only): 0 to 200 counts: beads here measure up to about 180 counts after treatment, so
# at the usual 0 to 4000 they would not show in fig0.
figMin=0
figMax=200
